function(kano_unattended_fixture runtime debug_crt)
    set(_target "kano_unattended_${runtime}")
    add_executable(${_target} unattended_fixture.cpp)
    target_link_libraries(${_target} PRIVATE KanoInfra::unattended)
    target_compile_definitions(${_target} PRIVATE NDEBUG)
    if(MSVC)
        if(debug_crt)
            target_compile_definitions(${_target} PRIVATE _DEBUG)
        else()
            target_compile_options(${_target} PRIVATE /U_DEBUG)
        endif()
        if(runtime MATCHES "^md")
            set(_runtime_library "MultiThreadedDLL")
        else()
            set(_runtime_library "MultiThreaded")
        endif()
        if(debug_crt)
            if(runtime MATCHES "^md")
                set(_runtime_library "MultiThreadedDebugDLL")
            else()
                set(_runtime_library "MultiThreadedDebug")
            endif()
        endif()
        set_property(TARGET ${_target} PROPERTY MSVC_RUNTIME_LIBRARY "${_runtime_library}")
        set(_dll "kano_unattended_dll_${runtime}")
        add_library(${_dll} SHARED unattended_crt_dll.cpp)
        target_link_libraries(${_dll} PRIVATE KanoInfra::unattended)
        if(debug_crt)
            target_compile_definitions(${_dll} PRIVATE _DEBUG)
            set_property(TARGET ${_dll} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDebug)
        else()
            target_compile_options(${_dll} PRIVATE /U_DEBUG)
            set_property(TARGET ${_dll} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
        endif()
        target_link_libraries(${_target} PRIVATE ${_dll})
    endif()

    foreach(_mode normal check assert abort crash)
        if(_mode STREQUAL normal)
            set(_diagnostic "normal completion")
        elseif(_mode STREQUAL check)
            set(_diagnostic "Always-active check failed")
        elseif(_mode STREQUAL assert)
            set(_diagnostic "assert")
        elseif(_mode STREQUAL abort)
            set(_diagnostic "Deliberate unattended abort")
        else()
            if(WIN32)
                set(_diagnostic "Fatal native exception")
            else()
                # POSIX retains fatal signal status; no replacement signal
                # handler fabricates a crash diagnostic.
                set(_diagnostic "")
            endif()
        endif()
        add_test(NAME "unattended.${runtime}.${_mode}"
            COMMAND kano_unattended_harness "$<TARGET_FILE:${_target}>" "${_mode}" "${_diagnostic}")
    endforeach()
    if(MSVC)
        foreach(_mode invalid dll-invalid)
            add_test(NAME "unattended.${runtime}.${_mode}"
                COMMAND kano_unattended_harness "$<TARGET_FILE:${_target}>" "${_mode}" "Fatal invalid CRT parameter")
        endforeach()
        if(debug_crt)
            foreach(_mode narrow wide dll-narrow dll-wide)
                add_test(NAME "unattended.${runtime}.${_mode}"
                    COMMAND kano_unattended_harness "$<TARGET_FILE:${_target}>" "${_mode}" "CRT report:")
            endforeach()
        endif()
    endif()

    foreach(_policy opt-out agent ci disabled unset)
        if(_policy STREQUAL opt-out)
            set(_environment "KANO_UNATTENDED=0;KANO_AGENT_MODE=1;CI=true")
            set(_expected 0)
        elseif(_policy STREQUAL agent)
            set(_environment "KANO_UNATTENDED=;KANO_AGENT_MODE=1;CI=")
            set(_expected 1)
        elseif(_policy STREQUAL ci)
            set(_environment "KANO_UNATTENDED=;KANO_AGENT_MODE=;CI=true")
            set(_expected 1)
        elseif(_policy STREQUAL disabled)
            set(_environment "KANO_UNATTENDED=OFF;KANO_AGENT_MODE=1;CI=true")
            set(_expected 0)
        else()
            set(_environment "KANO_UNATTENDED=;KANO_AGENT_MODE=;CI=")
            set(_expected 0)
        endif()
        add_test(NAME "unattended.${runtime}.environment.${_policy}" COMMAND ${_target} requested "${_expected}")
        set_property(TEST "unattended.${runtime}.environment.${_policy}" PROPERTY ENVIRONMENT "${_environment}")
    endforeach()
endfunction()
