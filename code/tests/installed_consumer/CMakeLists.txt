cmake_minimum_required(VERSION 3.16)
project(kano-infra-installed-consumer LANGUAGES CXX)
find_package(KanoInfra REQUIRED CONFIG COMPONENTS unattended scheduler process platform)
add_executable(installed_unattended main.cpp)
target_link_libraries(installed_unattended PRIVATE KanoInfra::unattended KanoInfra::scheduler KanoInfra::process KanoInfra::platform)
enable_testing()
if(WIN32)
    find_program(KANO_SCHEDULER_HELPER NAMES kano_task_scheduler
        HINTS "${KanoInfra_DIR}/../../../bin" NO_DEFAULT_PATH)
    if(NOT KANO_SCHEDULER_HELPER)
        message(FATAL_ERROR "Installed Scheduler helper is missing")
    endif()
    add_test(NAME installed.scheduler COMMAND installed_unattended "${KANO_SCHEDULER_HELPER}")
else()
    add_test(NAME installed.scheduler COMMAND installed_unattended)
endif()
set_tests_properties(installed.scheduler PROPERTIES TIMEOUT 30 ENVIRONMENT "KANO_UNATTENDED=1")
